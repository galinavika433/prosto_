# Задание 2. 
Пользователь вводит две границы диапазона, вывести на экран все числа из этого диапазона. Предусмотреть, чтобы пользователь мог вводить границы диапазона в произвольном порядке.
- вывести все четные числа из диапазона. 
- вывести все нечетные числа из диапазона. 
- вывести все числа, кратные семи.

```c++
#include <iostream>
using namespace std;

int main()
{
    int a, b;
    int i;

    cin >> a >> b;

    if (a > b)
    {
        int temp = a;
        a = b;
        b = temp;
    }

    cout << "все числа: ";

    i = a;
    while (i <= b)
    {
        cout << i << " ";
        i++;
    }

    cout << endl;

    cout << "четные числа: ";

    i = a;
    while (i <= b)
    {
        if (i % 2 == 0)
        {
            cout << i << " ";
        }
        i++;
    }

    cout << endl;

    cout << "нечетные числа: ";

    i = a;
    while (i <= b)
    {
        if (i % 2 != 0)
        {
            cout << i << " ";
        }
        i++;
    }

    cout << endl;

    cout << "числа кратные 7: ";

    i = a;
    while (i <= b)
    {
        if (i % 7 == 0)
        {
            cout << i << " ";
        }
        i++;
    }

    return 0;
}
```
